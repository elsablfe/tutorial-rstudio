Gràfic de barres (ELISA)

En aquest cas les dades sintètiques son de nivells de Igs en rentat intestinal. Tots estan en mg/mL menys les IgE que estan en ng/mL.

#llibreries
library(tidyverse) #conté 9 paquets d'edició i visualització 
library(here) #al treballar en un projecte
library(readxl) #per importar dades que estan en un fitxer d'excel
library(janitor) #ajudar a netejar les dades
library(skimr) #per obtenir un resum de les dades complert
library(broom) #poder fer servir funcions com tidy
library(rstatix) #funcions d'estadística

##Importar i preparar dades-----

raw_data <- read_excel(here("Basic", "data", "elisa.xlsx"))

data <- raw_data %>% 
  mutate(across(c(group, sex, rat, generation), as.factor))

longdata <- data %>% 
  pivot_longer(cols="IgA":"IgD", names_to="metabolite", values_to = "value" ) %>% 
  mutate(
    group = factor(group, 
                   levels = c("REF", "S", "PGS")), 
    generation = factor(generation, 
                        levels = c( "M", "Cd1", "Cd21", "CA")),
    metabolite = factor(metabolite,
                        levels = c("IgA", "IgG", "IgE", "IgM", "IgD")))

Gràfic de barres amb punts comú

Preparem tot el que vulguem personalitzar. En aquest cas el metabolite_name us ho mostro per que sapigueu que si voleu canviar el nom ho podeu fer sense canviar el dataset.

theme_set(theme_classic()) 

colors <- c("REF" = "#999999",   # Grey
            "PGS" = "#1b9e77",   # Green 1
            "S" = "#66a61e",   # Green 2
            "G" = "#a6d854",   # Green 3
            "P" = "#d9f0a3")   # Green 4

generation_name <- c(
  M   = "Dams",
  Cd1 = "Day-1 offspring",
  Cd21= "Day-21 offspring",
  CA  = "Adult offspring")

metabolite_name <- c(
  IgA   = "Immonuglobulin type A",
  IgG = "Immonuglobulin type G",
  IgE= "Immonuglobulin type E",
  IgM = "Immonuglobulin type M",
  IgD= "Immonuglobulin type D")

Fem el gràfic

ggplot <- longdata %>% 
  na.omit() %>% 
  ggplot(aes(x=group, y=value, colour=group))+ 
  geom_jitter(alpha = 0.5)+ #gràfic punts
  scale_x_discrete(limits = c("REF", "S","PGS"), 
                   expand = expansion(add = 1))+ #afegir espai entre grups
  facet_grid(generation~metabolite, 
             labeller = labeller(metabolite = metabolite_name, generation = generation_name))+ #ordre 
  stat_summary(fun = mean, geom = "col", width = 0.7, fill=NA, linewidth = 0.6) +
  stat_summary(fun.data = mean_se, geom = "errorbar", width = 0.5, linewidth = 0.6) +
  scale_colour_manual(values=colors, 
                      breaks = c("REF", "S", "PGS"))+ 
  labs(title= "Gut wash", 
       x= "Group", 
       y= "Concentration",
       colour= "Group")+  
  theme(plot.title = element_text(hjust = 0.5), 
        strip.text.x = element_text(size = 12),
        strip.text.y = element_text(size = 12, hjust = 0.75),
        strip.background = element_blank(),
        panel.spacing = unit(0.5, "cm")) #espai entre els mini gràfics

Gràfic de barres amb punts per metabolit

En aquest cas no té tant sentit fer els gràfics conjunts per tots els metabolits ja que estan en diferents unitats, per tant us recomano fer-los per separat.

ggplot <- longdata %>% 
  na.omit() %>% 
  filter(metabolite == "IgA") %>% 
  ggplot(aes(x=group, y=value, colour=group))+ 
  geom_jitter(alpha = 0.5)+ #gràfic punts
  scale_x_discrete(limits = c("REF", "S","PGS"), 
                   expand = expansion(add = 1))+ #afegir espai entre grups
  scale_y_continuous(expand = expansion(mult = c(0, 0.05)), limits = c(0, NA))+
  facet_grid(. ~ generation, 
             labeller = labeller(generation = generation_name))+ #ordre 
  stat_summary(fun = mean, geom = "col", width = 0.7, fill=NA, linewidth = 0.6) +
  stat_summary(fun.data = mean_se, geom = "errorbar", width = 0.4, linewidth = 0.6) +
  scale_colour_manual(values=colors, 
                      breaks = c("REF", "S", "PGS"))+ 
  labs(title= "Gut wash", 
       x= "Group", 
       y= "[IgA] (mg/mL)",
       colour= "Group")+  
  theme(plot.title = element_text(hjust = 0.5), 
        strip.text.x = element_text(size = 12),
        strip.text.y = element_text(size = 12, hjust = 0.75),
        strip.background = element_blank(),
        panel.spacing = unit(0.5, "cm"),
        aspect.ratio = 1.75)

Gràfic de barres amb punts separat per sexe

També podem separar dins del gràfic per sexe, i fer una barra per cada.

ggplot <- longdata %>% 
  na.omit() %>% 
  filter(metabolite == "IgA") %>% 
  group_by(sex) %>% 
  ggplot(aes(x=group, y=value, colour=sex))+ 
  geom_jitter(alpha = 0.5, position = position_jitterdodge(
      dodge.width = 0.75,   # separació entre sexes
      jitter.width = 0.25   # dispersió horitzontal real
  ))+ #gràfic punts
  scale_x_discrete(limits = c("REF", "S","PGS"), 
                   expand = expansion(add = 1))+ #afegir espai entre grups
  scale_y_continuous(expand = expansion(mult = c(0, 0.05)), limits = c(0, NA))+
  facet_grid(. ~ generation, 
             labeller = labeller(generation = generation_name))+ #ordre 
  stat_summary(fun = mean, geom = "col", width = 0.7, fill=NA, linewidth = 0.6, position = position_dodge(width = 0.75)) +
  stat_summary(fun.data = mean_se, geom = "errorbar", width = 0.4, linewidth = 0.6, position = position_dodge(width = 0.75)) +
  scale_colour_manual(values= c("pink", "lightblue", "grey"))+
  scale_fill_manual(values= c("pink", "lightblue", "grey"))+
  labs(title= "Gut wash", 
       x= "Group", 
       y= "[IgA] (mg/mL)",
       colour= "Sex")+  
  theme(plot.title = element_text(hjust = 0.5), 
        strip.text.x = element_text(size = 12),
        strip.text.y = element_text(size = 12, hjust = 0.75),
        strip.background = element_blank(),
        panel.spacing = unit(0.5, "cm"),
        aspect.ratio = 1.75)

Preparar i afegir significància

posthoc_dunn <- longdata %>%
  group_by(metabolite) %>%
  dunn_test(
    value ~ generation,
    p.adjust.method = "bonferroni")

gen_levels <- c("M", "Cd1", "Cd21", "CA")

posthoc_graph <- posthoc_dunn %>% 
  filter(metabolite == "IgE") %>% 
  add_xy_position (x= "generation") %>% 
  filter(p.adj < 0.05) %>%
  mutate(
    x1 = match(group1, gen_levels),
    x2 = match(group2, gen_levels),
    bracket_length = abs(x2 - x1)) %>% 
  arrange(bracket_length) %>%
  mutate(
    y.min = max(longdata$value[longdata$metabolite == "IgE"], na.rm = TRUE),
    y.position = y.min + 10 + seq(1, by = 20, length.out = n())) #by = 20 perque quedi mes separat

library(ggpubr)
ggplot + 
  stat_pvalue_manual(
    posthoc_graph,
    tip.length = 0.025,
    bracket.nudge.y = -1, 
    hide.ns = TRUE)
  • y.min = max(longdata$value[longdata$metabolite == "NGF"], na.rm = TRUE), Aquesta línia de codi el que ens indica és que comencin els brackets al valor màxim.

Gràfic diferents colors punts i barres

Per a posar de diferent color els punts i les barres (en el meu cas totalment exploratori), el que s’ha de fer és eliminar l’estètica principal de color i posar-la a cada element que vulguis que tingui color, indicant d’on ha de treure els valors en cada cas.

ggplot <- longdata %>% 
  filter(metabolite == "IgE") %>% 
  na.omit() %>% 
  ggplot(aes(x=generation, y=value))+ 
  geom_jitter(aes(colour=group),alpha = 0.5, width = 0.25)+ 
  scale_x_discrete(limits = c("M", "Cd1","Cd21", "CA"), 
                   expand = expansion(add = 1), 
                   labels=generation_name)+ #aqui tambe hem canviat com apareixen al gràfic
  scale_y_continuous(expand = expansion(mult = c(0, 0.05)), limits = c(0, NA))+
  stat_summary(aes(colour = generation),fun = mean, geom = "col", width = 0.7, fill=NA, linewidth = 0.6) +
  stat_summary(aes(colour = generation),fun.data = mean_se, geom = "errorbar", width = 0.4, linewidth = 0.6) +
  scale_colour_manual(values=c(colors_group, colors_generation), 
                      labels=generation_name)+ 
  labs(title= "Gut wash", 
       x= "Generation", 
       y= "[IgE] (ng/mL)",
       colour= "Legend")+  
  theme(plot.title = element_text(hjust = 0.5), 
        strip.text.x = element_text(size = 12),
        axis.text.x = element_text(angle = 45, hjust = 1),
        strip.text.y = element_text(size = 12, hjust = 0.75),
        strip.background = element_blank(),
        panel.spacing = unit(0.5, "cm"),
        aspect.ratio = 1.75)
